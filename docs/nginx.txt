


server {

    listen 9000;
    server_name localhost;

    location / {
        root /Users/rjha/code/github/www;
        index index.html;
        autoindex on;
        # Show exact file sizes and local file times
        autoindex_exact_size off; # Displays file sizes as MB/KB instead of raw bytes
        autoindex_localtime on;   # Uses your Mac's local timezone instead of UTC
    }

    # proxy pass to FastAPI
    location /game {
        proxy_pass http://127.0.0.1:8000/game ;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
